function getStatusBadgeHtml(status) { const st = status || 'Pending'; if (st === 'Approved') { return `Approved`; } if (st === 'Rejected') { return `Rejected`; } return `Pending`; } function renderDepositHistory() { const user = getCurrentUser(); const list = document.getElementById('depositHistoryList'); const items = (user && user.depositHistory) || []; if (items.length === 0) { list.innerHTML = '
কোনো ডিপোজিট হিস্ট্রি পাওয়া যায়নি।
'; return; } list.innerHTML = items.map(item => `

${item.method} — ৳ ${item.amount}

TrxID: ${item.trx}

${item.time}

${getStatusBadgeHtml(item.status)}
${item.screenshot ? `` : ''}
`).join(''); } function renderWithdrawHistory() { const user = getCurrentUser(); const list = document.getElementById('withdrawHistoryList'); const items = (user && user.withdrawHistory) || []; if (items.length === 0) { list.innerHTML = '
কোনো উত্তোলন হিস্ট্রি পাওয়া যায়নি।
'; return; } list.innerHTML = items.map(item => `

${item.method} — ৳ ${item.amount}

নম্বর: ${item.number}

${item.time}

${getStatusBadgeHtml(item.status)}
`).join(''); }
